-- 2. TABLE PRODUCTS (catalogue)

  img         text,
  stock       integer not null default 1,
  featured    boolean not null default false,

-- 3. TABLE ORDERS (commandes)
create table if not exists orders (
  id              bigint generated by default as identity primary key,
  ref             text unique,
  total           numeric(10,2) not null default 0,
  status          text not null default 'Confirmé',
  customer_email  text,
  created_at      timestamptz default now()
);

-- 4. SÉCURITÉ (Row Level Security)
-- Accès public en lecture (le site affiche les produits sans auth)
alter table config   enable row level security;
alter table products enable row level security;
alter table orders   enable row level security;
 
-- config : lecture publique, écriture via anon (pour l'admin côté client)
create policy "config_read"   on config   for select using (true);
create policy "config_write"  on config   for all    using (true);
 
-- products : lecture publique, écriture via anon
create policy "products_read"  on products for select using (true);
create policy "products_write" on products for all    using (true);
 
-- orders : lecture + écriture via anon (pour passer les commandes)
create policy "orders_read"   on orders   for select using (true);
create policy "orders_write"  on orders   for insert using (true);
create policy "orders_admin"  on orders   for all    using (true);
